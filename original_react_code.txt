// --- 3. MAIN APP ---
export default function App() {
  const containerRef = useRef(null);
  const [mounted, setMounted] = useState(false);
  
  // State
  const [schema, setSchema] = useState(INITIAL_SCHEMA);
  const [activeViewKey, setActiveViewKey] = useState('EVENT_STORMING');
  
  // Editor / Selection State
  const [activeTab, setActiveTab] = useState('entities'); 
  const [selectedId, setSelectedId] = useState(null); 
  
  // Canvas Viewport State
  const [transform, setTransform] = useState({ x: 50, y: 100, scale: 0.45 });
  const transformRef = useRef(transform);
  useEffect(() => { transformRef.current = transform; }, [transform]);
  
  // Presentation & Interaction State
  const [activeStep, setActiveStep] = useState(null);
  const [hoveredStack, setHoveredStack] = useState(null); // Maps to a Group/Stack for tooltips
  const [tooltipPos, setTooltipPos] = useState({ x: 0, y: 0 });

  const activeView = schema.views[activeViewKey];

  useEffect(() => {
    setMounted(false);
    setActiveStep(null);
    setHoveredStack(null);
    setTransform({ x: 50, y: window.innerWidth < 768 ? 150 : 100, scale: window.innerWidth < 768 ? 0.3 : 0.45 });
    setTimeout(() => setMounted(true), 100);
  }, [activeViewKey]);

  const [isPanning, setIsPanning] = useState(false);
  const [panStart, setPanStart] = useState({ x: 0, y: 0 });
  const [draggingNode, setDraggingNode] = useState(null); 

  // --- CAMERA ENGINE ---
  const animateTo = useCallback((targetX, targetY, targetScale) => {
    const start = transformRef.current;
    const duration = 650; const startTime = performance.now();
    const animate = (time) => {
      const p = Math.min((time - startTime) / duration, 1);
      const ease = 1 - Math.pow(1 - p, 3);
      setTransform({ x: start.x + (targetX - start.x) * ease, y: start.y + (targetY - start.y) * ease, scale: start.scale + (targetScale - start.scale) * ease });
      if (p < 1) requestAnimationFrame(animate);
    };
    requestAnimationFrame(animate);
  }, []);

  const focusOnNodes = useCallback((nodeIds) => {
    if (!containerRef.current || !nodeIds || nodeIds.length === 0) return;
    let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
    nodeIds.forEach(id => {
      const node = activeView.nodes.find(n => n.id === id);
      if (!node) return;
      const type = schema.entities[id]?.viewTypes[activeViewKey];
      const isSmall = [TYPES.USER, TYPES.EXTERNAL, TYPES.HOTSPOT, TYPES.DECISION].includes(type);
      minX = Math.min(minX, node.x); minY = Math.min(minY, node.y);
      maxX = Math.max(maxX, node.x + (isSmall ? SMALL_W : NODE_W)); maxY = Math.max(maxY, node.y + (isSmall ? SMALL_H : NODE_H));
    });
    if (minX === Infinity) return;
    
    // Sidebar on right, Stepper on bottom padding constraints
    const isSidebarOpen = selectedId !== null;
    const viewportW = containerRef.current.clientWidth - (isSidebarOpen ? 320 : 0); 
    const viewportH = containerRef.current.clientHeight - 240; 
    const padding = Math.min(viewportW * 0.1, 80); 
    
    const targetScale = Math.min((viewportW - padding * 2) / (maxX - minX + 100), (viewportH - padding * 2) / (maxY - minY + 100), 1.4); 
    const targetX = (viewportW / 2) - (((minX - 50) + (maxX - minX + 100) / 2) * targetScale);
    const targetY = (viewportH / 2) - (((minY - 50) + (maxY - minY + 100) / 2) * targetScale);
    animateTo(targetX, targetY, targetScale);
  }, [animateTo, activeView, activeViewKey, schema.entities, selectedId]);

  // --- CANVAS HANDLERS ---
  const handleWheel = useCallback((e) => {
    e.preventDefault();
    if (e.ctrlKey || e.metaKey) {
      const scaleBy = 1.05;
      const newScale = Math.min(Math.max(0.1, e.deltaY < 0 ? transformRef.current.scale * scaleBy : transformRef.current.scale / scaleBy), 3);
      if (containerRef.current) {
        const r = containerRef.current.getBoundingClientRect();
        setTransform(prev => ({
          x: (e.clientX - r.left) - ((e.clientX - r.left) - prev.x) * (newScale / prev.scale), 
          y: (e.clientY - r.top) - ((e.clientY - r.top) - prev.y) * (newScale / prev.scale), 
          scale: newScale 
        }));
      }
    } else { setTransform(prev => ({ ...prev, x: prev.x - e.deltaX, y: prev.y - e.deltaY })); }
  }, []);

  useEffect(() => {
    const el = containerRef.current;
    if (el) el.addEventListener('wheel', handleWheel, { passive: false });
    return () => { if (el) el.removeEventListener('wheel', handleWheel); };
  }, [handleWheel]);

  const handlePointerDown = (e) => {
    const targetNode = e.target.closest('.canvas-node');
    if (targetNode) {
      const id = targetNode.getAttribute('data-id');
      const node = activeView.nodes.find(n => n.id === id);
      if (node) {
        setDraggingNode({ id, offsetX: (e.clientX - transform.x) / transform.scale - node.x, offsetY: (e.clientY - transform.y) / transform.scale - node.y });
        setSelectedId(id);
        setActiveTab('entities');
        setActiveStep(null);
      }
      return;
    }
    
    // If not clicking a node, handle pan
    setIsPanning(true);
    setPanStart({ x: e.clientX - transform.x, y: e.clientY - transform.y });
    setSelectedId(null);
  };

  const handlePointerMove = (e) => {
    if (isPanning) {
      setTransform(prev => ({ ...prev, x: e.clientX - panStart.x, y: e.clientY - panStart.y }));
    } else if (draggingNode) {
      const newX = (e.clientX - transform.x) / transform.scale - draggingNode.offsetX;
      const newY = (e.clientY - transform.y) / transform.scale - draggingNode.offsetY;
      
      setSchema(prev => {
        const next = JSON.parse(JSON.stringify(prev)); 
        const node = next.views[activeViewKey].nodes.find(n => n.id === draggingNode.id);
        if (node) { node.x = Math.round(newX / 10) * 10; node.y = Math.round(newY / 10) * 10; } 
        return next;
      });
    }

    if (hoveredStack && !isDragging && !activeStep) {
      setTooltipPos({ x: e.clientX, y: e.clientY });
    }
  };

  const handlePointerUp = () => { setIsPanning(false); setDraggingNode(null); };

  // --- HOVER LOGIC ---
  const handleGroupPointerEnter = (e, group) => {
    if (activeStep || isDragging) return;
    if (group.desc) {
      setHoveredStack(group);
      setTooltipPos({ x: e.clientX, y: e.clientY });
    }
  };

  const handleGroupPointerLeave = () => {
    setHoveredStack(null);
  };

  // Determine the active nodes to highlight from step
  const activeNodeIds = activeStep ? activeStep.nodeIds : null;

  // --- MUTATION HELPERS (SIDEBAR) ---
  const updateEntity = (id, field, value) => {
    setSchema(prev => {
      const next = JSON.parse(JSON.stringify(prev));
      if (!next.entities[id]) return prev;
      if (field.startsWith('viewTypes.')) {
        const vKey = field.split('.')[1];
        next.entities[id].viewTypes[vKey] = value;
        const viewNodes = next.views[vKey].nodes;
        const exists = viewNodes.some(n => n.id === id);
        if (value && !exists) viewNodes.push({ id, x: 100, y: 100 }); 
        else if (!value && exists) next.views[vKey].nodes = viewNodes.filter(n => n.id !== id);
      } else {
        next.entities[id][field] = value;
      }
      return next;
    });
  };

  const addEntity = () => {
    const id = `ent_${Date.now()}`;
    setSchema(prev => {
      const next = JSON.parse(JSON.stringify(prev));
      next.entities[id] = { title: 'New Entity', desc: '', viewTypes: { [activeViewKey]: TYPES.EVENT } };
      next.views[activeViewKey].nodes.push({ id, x: 200, y: 200 });
      return next;
    });
    setSelectedId(id);
    setActiveTab('entities');
  };

  const deleteEntity = (id) => {
    setSchema(prev => {
      const next = JSON.parse(JSON.stringify(prev));
      delete next.entities[id];
      Object.keys(next.views).forEach(vKey => {
        next.views[vKey].nodes = next.views[vKey].nodes.filter(n => n.id !== id);
        next.views[vKey].groups.forEach(g => g.nodeIds = g.nodeIds ? g.nodeIds.filter(nid => nid !== id) : []);
        next.views[vKey].steps.forEach(s => {
           if(s.nodeIds) s.nodeIds = s.nodeIds.filter(nid => nid !== id);
           if(s.branches) s.branches.forEach(b => b.nodeIds = b.nodeIds.filter(nid => nid !== id));
        });
      });
      next.relations = next.relations.filter(r => r.from !== id && r.to !== id);
      return next;
    });
    setSelectedId(null);
  };

  const handleStepClick = (step) => {
    if (activeStep?.id === step.id) { 
      setActiveStep(null); 
      animateTo(50, 100, 0.45); 
    } else { 
      setActiveStep(step); 
      setHoveredStack(null); 
      focusOnNodes(step.nodeIds);
      
      // Auto-scroll the step pane to keep active step centered
      setTimeout(() => {
        const el = document.getElementById(`step-card-${step.id}`);
        if (el) el.scrollIntoView({ behavior: 'smooth', inline: 'center', block: 'nearest' });
      }, 50);
    }
  };

  // --- RENDER SIDEBAR ---
  const renderSidebar = () => {
    return (
      <div className={`absolute top-0 right-0 bottom-0 w-80 bg-white border-l border-slate-200 flex flex-col h-full z-40 shadow-2xl transition-transform duration-300 ease-in-out ${selectedId && !activeStep ? 'translate-x-0' : 'translate-x-full'}`}>
        <div className="flex bg-slate-100 p-2 gap-1 border-b border-slate-200 shrink-0">
          {[
            { id: 'entities', icon: Database, title: 'Entities' }, { id: 'relations', icon: Share2, title: 'Relations' },
            { id: 'groups', icon: LayoutTemplate, title: 'Groups' }, { id: 'steps', icon: Play, title: 'Steps' }
          ].map(t => (
            <button key={t.id} onClick={() => { setActiveTab(t.id); setSelectedId(null); }} title={t.title} className={`flex-1 flex justify-center py-2 rounded text-slate-600 transition-colors ${activeTab === t.id ? 'bg-white shadow-sm font-bold text-blue-600' : 'hover:bg-slate-200'}`}>
              <t.icon size={16} />
            </button>
          ))}
          <button onClick={() => setSelectedId(null)} className="flex items-center justify-center px-3 hover:bg-slate-200 rounded text-slate-400 hover:text-slate-700 transition"><X size={16}/></button>
        </div>

        <div className="flex-1 overflow-y-auto p-4">
          
          {/* ENTITIES TAB */}
          {activeTab === 'entities' && (
            <div>
              <div className="flex justify-between items-center mb-4">
                <h2 className="font-black text-slate-800 uppercase tracking-wider text-xs">Domain Entities</h2>
                <button onClick={addEntity} className="bg-blue-100 hover:bg-blue-200 text-blue-700 p-1.5 rounded"><Plus size={14}/></button>
              </div>
              
              {!selectedId || !schema.entities[selectedId] ? (
                <div className="space-y-2">
                  {Object.entries(schema.entities).map(([id, ent]) => {
                    const isActiveInView = ent.viewTypes[activeViewKey];
                    return (
                      <div key={id} onClick={() => setSelectedId(id)} className={`p-3 rounded-lg border cursor-pointer hover:shadow-md transition-all ${isActiveInView ? 'bg-white border-slate-200' : 'bg-slate-50 border-slate-100 opacity-60'}`}>
                        <div className="font-bold text-sm text-slate-800">{ent.title}</div>
                        <div className="text-[10px] text-slate-500 mt-1">{isActiveInView || 'Not in view'}</div>
                      </div>
                    )
                  })}
                </div>
              ) : (
                <div className="space-y-4 animate-in fade-in slide-in-from-right-4">
                  <button onClick={() => setSelectedId('list')} className="text-xs text-blue-500 font-bold mb-2 flex items-center">&larr; Back to list</button>
                  <div>
                    <label className="block text-[10px] font-bold text-slate-500 uppercase mb-1">Title</label>
                    <input type="text" value={schema.entities[selectedId]?.title || ''} onChange={e => updateEntity(selectedId, 'title', e.target.value)} className="w-full border border-slate-300 rounded p-2 text-sm outline-none" />
                  </div>
                  <div>
                    <label className="block text-[10px] font-bold text-slate-500 uppercase mb-1">Description</label>
                    <textarea value={schema.entities[selectedId]?.desc || ''} onChange={e => updateEntity(selectedId, 'desc', e.target.value)} className="w-full border border-slate-300 rounded p-2 text-sm outline-none min-h-[80px]" />
                  </div>
                  
                  <div className="bg-slate-50 p-3 rounded-lg border border-slate-200 space-y-3">
                    <h3 className="text-[10px] font-bold text-slate-500 uppercase">View Projections</h3>
                    {Object.keys(schema.views).map(vKey => (
                      <div key={vKey}>
                        <label className="block text-xs text-slate-700 mb-1">{schema.views[vKey].name}</label>
                        <select 
                          value={schema.entities[selectedId]?.viewTypes[vKey] || ''} 
                          onChange={e => updateEntity(selectedId, `viewTypes.${vKey}`, e.target.value)}
                          className="w-full border border-slate-300 rounded p-1.5 text-xs bg-white outline-none"
                        >
                          <option value="">-- Exclude --</option>
                          {Object.values(TYPES).map(t => <option key={t} value={t}>{t}</option>)}
                        </select>
                      </div>
                    ))}
                  </div>
                  <button onClick={() => deleteEntity(selectedId)} className="w-full mt-4 py-2 bg-red-50 text-red-600 rounded flex items-center justify-center gap-2 text-sm font-bold hover:bg-red-100 transition"><Trash2 size={14}/> Delete Entity</button>
                </div>
              )}
            </div>
          )}

          {/* RELATIONS TAB */}
          {activeTab === 'relations' && (
            <div>
              <h2 className="font-black text-slate-800 uppercase tracking-wider text-xs mb-4">Connections (Active View)</h2>
              <div className="bg-blue-50 p-3 rounded-xl border border-blue-100 mb-6">
                <h3 className="text-xs font-bold text-blue-800 mb-2">Create Link</h3>
                <div className="flex gap-2 items-center mb-2">
                  <select id="relFrom" className="flex-1 text-xs p-1.5 rounded border border-blue-200 outline-none">
                    <option value="">Source...</option>
                    {activeView.nodes.map(n => <option key={n.id} value={n.id}>{schema.entities[n.id]?.title}</option>)}
                  </select>
                  <span className="text-blue-300">&rarr;</span>
                  <select id="relTo" className="flex-1 text-xs p-1.5 rounded border border-blue-200 outline-none">
                    <option value="">Target...</option>
                    {activeView.nodes.map(n => <option key={n.id} value={n.id}>{schema.entities[n.id]?.title}</option>)}
                  </select>
                </div>
                <button 
                  onClick={() => {
                    const f = document.getElementById('relFrom').value; const t = document.getElementById('relTo').value;
                    if(f && t && f !== t) {
                      setSchema(p => {
                        const n = JSON.parse(JSON.stringify(p));
                        n.relations.push({ id: `rel_${Date.now()}`, from: f, to: t, views: [activeViewKey] });
                        return n;
                      });
                    }
                  }}
                  className="w-full bg-blue-600 text-white text-xs font-bold py-1.5 rounded hover:bg-blue-700"
                >Add Connection</button>
              </div>

              <div className="space-y-2">
                {schema.relations.filter(r => r.views.includes(activeViewKey)).map(rel => (
                  <div key={rel.id} className="flex items-center justify-between bg-white border border-slate-200 p-2 rounded shadow-sm text-xs">
                    <div className="flex items-center gap-2 truncate">
                      <span className="truncate max-w-[90px]">{schema.entities[rel.from]?.title || '?'}</span>
                      <span className="text-slate-400">&rarr;</span>
                      <span className="truncate max-w-[90px]">{schema.entities[rel.to]?.title || '?'}</span>
                    </div>
                    <button onClick={() => setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.relations = n.relations.filter(r => r.id !== rel.id); return n;})} className="text-red-400 hover:text-red-600"><X size={14}/></button>
                  </div>
                ))}
              </div>
            </div>
          )}

          {/* GROUPS TAB */}
          {activeTab === 'groups' && (
            <div>
              <div className="flex justify-between items-center mb-4">
                <h2 className="font-black text-slate-800 uppercase tracking-wider text-xs">Domains & Swimlanes</h2>
                <button onClick={() => {
                  const id = `g_${Date.now()}`;
                  setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].groups.push({ id, title: 'New Group', desc: '', nodeIds: [], color: 'rgba(96, 165, 250, 0.1)', borderColor: '#3b82f6', textColor: '#1e40af', isLane: false, y: 100, h: 200 }); return n; });
                  setSelectedId(id);
                }} className="bg-blue-100 hover:bg-blue-200 text-blue-700 p-1.5 rounded"><Plus size={14}/></button>
              </div>

              {!selectedId || !activeView.groups.find(g=>g.id === selectedId) ? (
                <div className="space-y-2">
                  {activeView.groups.map(g => (
                    <div key={g.id} onClick={() => setSelectedId(g.id)} className="p-3 bg-white border border-slate-200 rounded-lg cursor-pointer hover:shadow-md flex items-center justify-between">
                      <div>
                        <div className="font-bold text-sm text-slate-800">{g.title}</div>
                        <div className="text-[10px] text-slate-500 mt-0.5">{g.isLane ? 'Swimlane' : `${g.nodeIds?.length || 0} nodes`}</div>
                      </div>
                      <div className="w-4 h-4 rounded-full border border-black/10" style={{ backgroundColor: g.borderColor || g.color }}></div>
                    </div>
                  ))}
                </div>
              ) : (() => {
                const group = activeView.groups.find(g => g.id === selectedId);
                return (
                  <div className="space-y-4 animate-in fade-in slide-in-from-right-4">
                    <button onClick={() => setSelectedId('list')} className="text-xs text-blue-500 font-bold mb-2 flex items-center">&larr; Back to list</button>
                    <div>
                      <label className="block text-[10px] font-bold text-slate-500 uppercase mb-1">Group Title</label>
                      <input type="text" value={group.title || ''} onChange={e => { setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].groups.find(g => g.id === selectedId).title = e.target.value; return n; }); }} className="w-full border border-slate-300 rounded p-2 text-sm outline-none" />
                    </div>
                    <div>
                      <label className="block text-[10px] font-bold text-slate-500 uppercase mb-1">Description (Tooltip)</label>
                      <textarea value={group.desc || ''} onChange={e => { setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].groups.find(g => g.id === selectedId).desc = e.target.value; return n; }); }} className="w-full border border-slate-300 rounded p-2 text-sm outline-none min-h-[60px]" />
                    </div>
                    <label className="flex items-center gap-2 text-sm">
                      <input type="checkbox" checked={group.isLane || false} onChange={e => { setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].groups.find(g => g.id === selectedId).isLane = e.target.checked; return n; }); }} /> Render as Swimlane
                    </label>

                    {!group.isLane && (
                      <div>
                        <label className="block text-[10px] font-bold text-slate-500 uppercase mb-2">Assigned Nodes</label>
                        <div className="max-h-48 overflow-y-auto border border-slate-200 rounded bg-slate-50 p-2 space-y-1">
                          {activeView.nodes.map(n => {
                            const isChecked = group.nodeIds?.includes(n.id);
                            return (
                              <label key={n.id} className="flex items-center gap-2 text-xs p-1 hover:bg-slate-200 rounded cursor-pointer select-none">
                                {isChecked ? <CheckSquare size={14} className="text-blue-500"/> : <Square size={14} className="text-slate-300"/>}
                                <span className="truncate">{schema.entities[n.id]?.title}</span>
                                <input type="checkbox" className="hidden" checked={isChecked} onChange={(e) => {
                                  setSchema(p => { 
                                    const nxt = JSON.parse(JSON.stringify(p)); 
                                    const g = nxt.views[activeViewKey].groups.find(gx => gx.id === selectedId);
                                    if(!g.nodeIds) g.nodeIds = [];
                                    if (e.target.checked) g.nodeIds.push(n.id); else g.nodeIds = g.nodeIds.filter(id => id !== n.id);
                                    return nxt; 
                                  });
                                }}/>
                              </label>
                            )
                          })}
                        </div>
                      </div>
                    )}
                    <button onClick={() => { setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].groups = n.views[activeViewKey].groups.filter(g => g.id !== selectedId); return n; }); setSelectedId('list'); }} className="w-full mt-4 py-2 bg-red-50 text-red-600 rounded flex items-center justify-center gap-2 text-sm font-bold hover:bg-red-100 transition"><Trash2 size={14}/> Delete Group</button>
                  </div>
                )
              })()}
            </div>
          )}

          {/* STEPS TAB */}
          {activeTab === 'steps' && (
            <div>
              <div className="flex justify-between items-center mb-4">
                <h2 className="font-black text-slate-800 uppercase tracking-wider text-xs">Guided Tour Steps</h2>
                <button onClick={() => {
                  const id = `s_${Date.now()}`;
                  setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].steps.push({ id, type: 'linear', title: 'New Step', reason: '', nodeIds: [] }); return n; });
                  setSelectedId(id);
                }} className="bg-blue-100 hover:bg-blue-200 text-blue-700 p-1.5 rounded"><Plus size={14}/></button>
              </div>

              {!selectedId || !activeView.steps.find(s=>s.id === selectedId) ? (
                <div className="space-y-2">
                  {activeView.steps.map((s, idx) => (
                    <div key={s.id} onClick={() => setSelectedId(s.id)} className="p-3 bg-white border border-slate-200 rounded-lg cursor-pointer hover:shadow-md relative overflow-hidden">
                      {s.type === 'branch' && <div className="absolute top-0 right-0 bg-blue-100 text-blue-800 text-[9px] font-bold px-2 py-0.5 rounded-bl">BRANCH</div>}
                      <div className="text-[10px] font-bold text-blue-500 uppercase mb-1">Step {idx + 1}</div>
                      <div className="font-bold text-sm text-slate-800 truncate">{s.title}</div>
                      <div className="text-xs text-slate-500 mt-1 truncate">{s.reason || 'No description'}</div>
                    </div>
                  ))}
                </div>
              ) : (() => {
                const step = activeView.steps.find(s => s.id === selectedId);
                
                if (step.type === 'branch') {
                  return (
                    <div className="space-y-4 animate-in fade-in slide-in-from-right-4">
                       <button onClick={() => setSelectedId('list')} className="text-xs text-blue-500 font-bold mb-2 flex items-center">&larr; Back to list</button>
                       <div className="p-4 bg-amber-50 text-amber-800 rounded border border-amber-200 text-sm">
                         Complex <b>Branching Steps</b> must be edited directly in the JSON Schema in this version. 
                       </div>
                       <button onClick={() => { setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].steps = n.views[activeViewKey].steps.filter(sx => sx.id !== selectedId); return n; }); setSelectedId('list'); }} className="w-full mt-4 py-2 bg-red-50 text-red-600 rounded flex items-center justify-center gap-2 text-sm font-bold hover:bg-red-100 transition"><Trash2 size={14}/> Delete Step</button>
                    </div>
                  );
                }

                return (
                  <div className="space-y-4 animate-in fade-in slide-in-from-right-4">
                    <button onClick={() => setSelectedId('list')} className="text-xs text-blue-500 font-bold mb-2 flex items-center">&larr; Back to list</button>
                    
                    <div>
                      <label className="block text-[10px] font-bold text-slate-500 uppercase mb-1">Step Title</label>
                      <input type="text" value={step.title} onChange={e => { setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].steps.find(sx => sx.id === selectedId).title = e.target.value; return n; }); }} className="w-full border border-slate-300 rounded p-2 text-sm outline-none" />
                    </div>
                    <div>
                      <label className="block text-[10px] font-bold text-slate-500 uppercase mb-1">Description / Reason</label>
                      <textarea value={step.reason} onChange={e => { setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].steps.find(sx => sx.id === selectedId).reason = e.target.value; return n; }); }} className="w-full border border-slate-300 rounded p-2 text-sm outline-none min-h-[80px]" />
                    </div>

                    <div>
                      <label className="block text-[10px] font-bold text-slate-500 uppercase mb-2">Focus Nodes</label>
                      <div className="max-h-48 overflow-y-auto border border-slate-200 rounded bg-slate-50 p-2 space-y-1">
                        {activeView.nodes.map(n => {
                          const isChecked = step.nodeIds?.includes(n.id);
                          return (
                            <label key={n.id} className="flex items-center gap-2 text-xs p-1 hover:bg-slate-200 rounded cursor-pointer select-none">
                              {isChecked ? <CheckSquare size={14} className="text-blue-500"/> : <Square size={14} className="text-slate-300"/>}
                              <span className="truncate">{schema.entities[n.id]?.title}</span>
                              <input type="checkbox" className="hidden" checked={isChecked} onChange={(e) => {
                                setSchema(p => { 
                                  const nxt = JSON.parse(JSON.stringify(p)); 
                                  const s = nxt.views[activeViewKey].steps.find(sx => sx.id === selectedId);
                                  if(!s.nodeIds) s.nodeIds = [];
                                  if (e.target.checked) s.nodeIds.push(n.id); else s.nodeIds = s.nodeIds.filter(id => id !== n.id);
                                  return nxt; 
                                });
                              }}/>
                            </label>
                          )
                        })}
                      </div>
                    </div>
                    <button onClick={() => { setSchema(p => { const n = JSON.parse(JSON.stringify(p)); n.views[activeViewKey].steps = n.views[activeViewKey].steps.filter(sx => sx.id !== selectedId); return n; }); setSelectedId('list'); }} className="w-full mt-4 py-2 bg-red-50 text-red-600 rounded flex items-center justify-center gap-2 text-sm font-bold hover:bg-red-100 transition"><Trash2 size={14}/> Delete Step</button>
                  </div>
                )
              })()}
            </div>
          )}

        </div>
      </div>
    );
  };

  return (
    <div className="flex h-screen w-screen bg-slate-50 overflow-hidden font-sans text-slate-800 touch-none select-none relative">
      
      {/* Global CSS Animations */}
      <style>{`
        .hide-scrollbar::-webkit-scrollbar { display: none; }
        @keyframes dash-flow { from { stroke-dashoffset: 16; } to { stroke-dashoffset: 0; } }
        .edge-flow { stroke-dasharray: 8 8; animation: dash-flow 1.5s linear infinite; }
        .edge-flow-fast { stroke-dasharray: 8 8; animation: dash-flow 0.4s linear infinite; }
        @keyframes pulse-soft { 0%, 100% { opacity: 0.7; } 50% { opacity: 0.3; } }
        .animate-pulse-soft { animation: pulse-soft 4s ease-in-out infinite; }
        @keyframes alert-pulse { 0%, 100% { filter: drop-shadow(0 0 4px rgba(239, 68, 68, 0.4)); } 50% { filter: drop-shadow(0 0 12px rgba(239, 68, 68, 0.8)); } }
        .hotspot-pulse { animation: alert-pulse 2s ease-in-out infinite; }
        @keyframes icon-zap { 0%, 100% { transform: scale(1) rotate(0deg); } 10%, 30% { transform: scale(1.2) rotate(15deg); } 20%, 40% { transform: scale(1.2) rotate(-15deg); } 50% { transform: scale(1) rotate(0deg); } }
        .anim-icon-zap { animation: icon-zap 3s ease-in-out infinite; }
        @keyframes icon-blink { 0%, 100% { opacity: 1; } 50% { opacity: 0.2; } }
        .anim-icon-blink { animation: icon-blink 1.5s steps(2, start) infinite; }
        @keyframes icon-float-heavy { 0%, 100% { transform: translateY(0); } 50% { transform: translateY(-3px); } }
        .anim-icon-float-heavy { animation: icon-float-heavy 3s ease-in-out infinite; }
        @keyframes icon-float { 0%, 100% { transform: translateY(0); } 50% { transform: translateY(-6px); } }
        .anim-icon-float { animation: icon-float 3s ease-in-out infinite; }
        @keyframes icon-pulse { 0%, 100% { transform: scale(1); } 50% { transform: scale(1.15); } }
        .anim-icon-pulse { animation: icon-pulse 2s ease-in-out infinite; }
        @keyframes icon-scan { 0%, 100% { transform: translateX(-2px); } 50% { transform: translateX(2px); } }
        .anim-icon-scan { animation: icon-scan 3s ease-in-out infinite; }
        @keyframes icon-wobble { 0%, 100% { transform: rotate(0deg); } 25% { transform: rotate(-8deg); } 75% { transform: rotate(8deg); } }
        .anim-icon-wobble { animation: icon-wobble 4s ease-in-out infinite; }
        @keyframes icon-drift { 0%, 100% { transform: translate(0, 0); } 50% { transform: translate(2px, -1.5px); } }
        .anim-icon-drift { animation: icon-drift 5s ease-in-out infinite; }
        @keyframes icon-ring { 0%, 100% { transform: rotate(0deg); } 25% { transform: rotate(-12deg); } 75% { transform: rotate(12deg); } }
        .anim-icon-ring { animation: icon-ring 0.6s ease-in-out infinite; }
        @keyframes icon-spin-slow { 100% { transform: rotate(360deg); } }
        .anim-icon-spin-slow { animation: icon-spin-slow 8s linear infinite; }
      `}</style>

      {/* --- SVG CANVAS AREA --- */}
      <div 
        ref={containerRef}
        className="flex-1 relative overflow-hidden bg-slate-50"
        onPointerDown={handlePointerDown} 
        onPointerMove={handlePointerMove} 
        onPointerUp={handlePointerUp}
        onPointerLeave={handlePointerUp}
        style={{ cursor: isPanning ? 'grabbing' : (draggingNode ? 'grabbing' : 'grab') }}
      >
        
        {/* --- VIEW SWITCHER --- */}
        <div className="absolute top-4 left-4 z-20 flex flex-col gap-2 pointer-events-none">
          <div className="bg-white/95 backdrop-blur-md p-1.5 rounded-xl shadow-lg border border-slate-200 flex gap-1 pointer-events-auto">
            {Object.keys(schema.views).map(key => {
              const viewConfig = schema.views[key];
              const ViewIcon = DYNAMIC_ICONS[viewConfig.icon] || Component;
              return (
                <button 
                  key={key} 
                  onClick={() => { setActiveViewKey(key); setSelectedId(null); setActiveStep(null); }}
                  className={`px-3 py-1.5 rounded-lg text-sm font-bold transition-all flex items-center gap-2 ${activeViewKey === key ? 'bg-slate-800 text-white shadow-md' : 'text-slate-600 hover:bg-slate-100'}`}
                >
                  <ViewIcon size={14}/>
                  {viewConfig.name}
                </button>
              )
            })}
          </div>
        </div>

        {/* --- ZOOM CONTROLS --- */}
        <div className={`absolute top-4 transition-all duration-300 flex gap-1 p-1 bg-white/95 backdrop-blur-md rounded-full shadow-lg border border-slate-200 z-20 pointer-events-auto ${selectedId && !activeStep ? 'right-[336px]' : 'right-4'}`}>
          <button onClick={() => animateTo(transform.x, transform.y, Math.max(transform.scale / 1.3, 0.1))} className="p-2 hover:bg-slate-100 rounded-full"><ZoomOut size={16}/></button>
          <div className="flex items-center justify-center w-10 text-xs font-bold text-slate-500">{Math.round(transform.scale * 100)}%</div>
          <button onClick={() => animateTo(transform.x, transform.y, Math.min(transform.scale * 1.3, 3))} className="p-2 hover:bg-slate-100 rounded-full"><ZoomIn size={16}/></button>
          <div className="w-px h-5 bg-slate-300 self-center mx-1"></div>
          <button onClick={() => { setActiveStep(null); animateTo(50, 100, 0.45); }} className="p-2 hover:bg-slate-100 rounded-full"><Locate size={16}/></button>
        </div>

        {/* --- DYNAMIC HOVER TOOLTIP (STACK/GROUP) --- */}
        {hoveredStack && !activeStep && !isDragging && (
          <div 
            className="absolute z-50 bg-slate-900 text-white p-4 rounded-xl shadow-2xl border border-slate-700 max-w-[280px] pointer-events-none animate-in fade-in"
            style={{ 
              left: tooltipPos.x + 20, 
              top: tooltipPos.y + 20, 
              transform: tooltipPos.x > window.innerWidth - 300 ? 'translateX(-100%)' : 'none' 
            }}
          >
            <div className="flex items-center gap-2 text-blue-400 mb-2">
              <Lightbulb size={18} />
              <span className="font-bold text-sm tracking-wide uppercase">{hoveredStack.title}</span>
            </div>
            <p className="text-slate-300 text-xs leading-relaxed">{hoveredStack.desc}</p>
          </div>
        )}

        {/* --- STEPPER PANEL (BOTTOM) --- */}
        {activeView.steps && activeView.steps.length > 0 && (
          <div className={`absolute bottom-4 left-0 transition-all duration-300 z-30 pointer-events-none flex flex-col items-center ${selectedId && !activeStep ? 'right-[320px]' : 'right-0'}`}>
            {activeStep && (
              <button 
                onClick={() => { setActiveStep(null); animateTo(50, 100, 0.45); }}
                className="mb-3 px-4 py-1.5 bg-slate-800 text-white rounded-full text-xs font-bold shadow-lg pointer-events-auto hover:bg-slate-700 transition flex items-center gap-2"
              >
                Clear Focus <X size={14} />
              </button>
            )}
            
            <div className="w-full flex gap-4 overflow-x-auto pb-4 pt-2 items-center px-6 pointer-events-auto hide-scrollbar scroll-smooth">
              {activeView.steps.map((step, idx) => {
                if (step.type === 'linear') {
                  const isActive = activeStep?.id === step.id;
                  return (
                    <div 
                      id={`step-card-${step.id}`}
                      key={step.id} onClick={() => handleStepClick(step)}
                      className={`flex-none w-64 p-3 rounded-xl cursor-pointer border transition-all duration-300 shadow-sm
                        ${isActive ? 'bg-blue-50 border-blue-400 shadow-md ring-2 ring-blue-200 ring-offset-2 scale-105' : 'bg-white/95 backdrop-blur border-slate-200 hover:border-blue-300 hover:shadow-md'}`}
                    >
                      <div className="flex items-center justify-between mb-2">
                        <span className={`text-[9px] font-extrabold uppercase tracking-widest px-2 py-0.5 rounded-sm ${isActive ? 'bg-blue-200 text-blue-800' : 'bg-slate-100 text-slate-500'}`}>Phase {idx + 1}</span>
                        {isActive && <Play size={12} className="text-blue-500 fill-blue-500 animate-pulse" />}
                      </div>
                      <h3 className={`text-sm font-bold mb-1 leading-tight ${isActive ? 'text-blue-900' : 'text-slate-800'}`}>{step.title}</h3>
                      <p className="text-[10px] text-slate-500 line-clamp-2 leading-relaxed">{step.reason}</p>
                    </div>
                  );
                } 
                if (step.type === 'branch') {
                  return (
                    <div key={step.id} className="flex-none flex flex-col gap-2 pl-5 ml-2 border-l-2 border-dashed border-slate-400 relative">
                      <div className="absolute -left-[11px] top-1/2 bg-slate-100 border-2 border-slate-400 text-slate-500 p-0.5 rounded-full -translate-y-1/2">
                        <GitBranch size={12} />
                      </div>
                      {step.branches.map(branch => {
                        const isActive = activeStep?.id === branch.id;
                        return (
                          <div 
                            id={`step-card-${branch.id}`}
                            key={branch.id} onClick={() => handleStepClick(branch)}
                            className={`w-56 p-2.5 rounded-lg cursor-pointer border transition-all duration-300 shadow-sm
                              ${isActive ? 'bg-blue-50 border-blue-400 shadow-md ring-2 ring-blue-200 ring-offset-1 scale-105 z-10' : 'bg-white/95 border-slate-200 hover:border-blue-300 hover:shadow-md'}`}
                          >
                            <div className="flex items-center justify-between mb-1">
                              <h3 className={`text-xs font-bold leading-tight ${isActive ? 'text-blue-900' : 'text-slate-800'}`}>{branch.title}</h3>
                              {isActive && <Play size={10} className="text-blue-500 fill-blue-500 animate-pulse" />}
                            </div>
                            <p className="text-[9px] text-slate-500 line-clamp-1">{branch.reason}</p>
                          </div>
                        );
                      })}
                    </div>
                  );
                }
                return null;
              })}
            </div>
          </div>
        )}

        {/* --- SVG RENDERER --- */}
        <svg width="100%" height="100%">
          <defs>
            <pattern id="dotGrid" width={20 * transform.scale} height={20 * transform.scale} patternUnits="userSpaceOnUse" patternTransform={`translate(${transform.x % (20 * transform.scale)}, ${transform.y % (20 * transform.scale)})`}>
              <circle cx="2" cy="2" r={1 * transform.scale} fill="#cbd5e1" />
            </pattern>
            <filter id="shadow" x="-20%" y="-20%" width="140%" height="140%"><feDropShadow dx="0" dy="4" stdDeviation="4" floodOpacity="0.1" /></filter>
            <filter id="glow" x="-20%" y="-20%" width="140%" height="140%"><feDropShadow dx="0" dy="0" stdDeviation="6" floodColor="#3b82f6" floodOpacity="0.6" /></filter>
            <marker id="arrowhead" markerWidth="10" markerHeight="7" refX="9" refY="3.5" orient="auto"><polygon points="0 0, 10 3.5, 0 7" fill="#94a3b8" /></marker>
            <marker id="arrowhead-highlight" markerWidth="10" markerHeight="7" refX="9" refY="3.5" orient="auto"><polygon points="0 0, 10 3.5, 0 7" fill="#3b82f6" /></marker>
          </defs>

          <rect width="100%" height="100%" fill="url(#dotGrid)" />

          <g transform={`translate(${transform.x}, ${transform.y}) scale(${transform.scale})`}>
            
            {/* GROUPS RENDERER */}
            {activeView.groups.map(group => {
              const isFaded = activeNodeIds !== null;
              
              if (group.isLane) {
                return (
                  <g 
                    key={group.id} 
                    style={{ transition: 'opacity 0.3s', opacity: isFaded ? 0.05 : 1 }}
                    onPointerEnter={(e) => handleGroupPointerEnter(e, group)}
                    onPointerLeave={handleGroupPointerLeave}
                  >
                    <rect x="-1000" y={group.y} width="5000" height={group.h} fill={group.color} opacity="0.3" stroke="#cbd5e1" strokeDasharray="10 5" />
                    <rect x="20" y={group.y} width="240" height="30" fill={group.color} opacity="0.8" rx="4" />
                    <text x="30" y={group.y + 20} fill="#334155" fontSize="14" fontWeight="bold" fontFamily="sans-serif">{group.title}</text>
                  </g>
                );
              }

              let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
              group.nodeIds?.forEach(id => {
                const node = activeView.nodes.find(n => n.id === id);
                if (!node) return;
                const type = schema.entities[id]?.viewTypes[activeViewKey];
                if (!type) return;
                const w = [TYPES.USER, TYPES.EXTERNAL, TYPES.HOTSPOT, TYPES.DECISION].includes(type) ? SMALL_W : NODE_W;
                const h = [TYPES.USER, TYPES.EXTERNAL, TYPES.HOTSPOT, TYPES.DECISION].includes(type) ? SMALL_H : NODE_H;
                minX = Math.min(minX, node.x); minY = Math.min(minY, node.y);
                maxX = Math.max(maxX, node.x + w); maxY = Math.max(maxY, node.y + h);
              });
              if (minX === Infinity) return null;
              const padding = 40;
              return (
                <g 
                  key={group.id} 
                  transform={`translate(${minX - padding}, ${minY - padding - 40})`} 
                  className="animate-pulse-soft" 
                  style={{ transition: 'opacity 0.3s', opacity: isFaded ? 0.05 : 1 }}
                  onPointerEnter={(e) => handleGroupPointerEnter(e, group)}
                  onPointerLeave={handleGroupPointerLeave}
                >
                  <rect width={(maxX - minX) + padding * 2} height={(maxY - minY) + padding * 2 + 40} fill={group.color} stroke={group.borderColor} strokeWidth="2" strokeDasharray="8 8" rx="16" />
                  <rect x="0" y="0" width={(maxX - minX) + padding * 2} height="40" fill={group.color} opacity="0.8" rx="16" />
                  <text x="20" y="25" fill={group.textColor} fontSize="16" fontWeight="800" fontFamily="sans-serif" style={{ pointerEvents: 'none' }}>{group.title}</text>
                </g>
              );
            })}

            {/* EDGES RENDERER */}
            {schema.relations.filter(r => r.views.includes(activeViewKey)).map((edge, idx) => {
              const fromNode = activeView.nodes.find(n => n.id === edge.from);
              const toNode = activeView.nodes.find(n => n.id === edge.to);
              if (!fromNode || !toNode || !schema.entities[edge.from] || !schema.entities[edge.to]) return null;

              const fromType = schema.entities[edge.from].viewTypes[activeViewKey];
              const toType = schema.entities[edge.to].viewTypes[activeViewKey];

              const cx1 = fromNode.x + ([TYPES.USER, TYPES.EXTERNAL, TYPES.HOTSPOT, TYPES.DECISION].includes(fromType) ? SMALL_W : NODE_W) / 2;
              const cy1 = fromNode.y + ([TYPES.USER, TYPES.EXTERNAL, TYPES.HOTSPOT, TYPES.DECISION].includes(fromType) ? SMALL_H : NODE_H) / 2;
              const cx2 = toNode.x + ([TYPES.USER, TYPES.EXTERNAL, TYPES.HOTSPOT, TYPES.DECISION].includes(toType) ? SMALL_W : NODE_W) / 2;
              const cy2 = toNode.y + ([TYPES.USER, TYPES.EXTERNAL, TYPES.HOTSPOT, TYPES.DECISION].includes(toType) ? SMALL_H : NODE_H) / 2;

              const isVerticalJump = Math.abs(cy2 - cy1) > 100;
              const cp1x = isVerticalJump ? cx1 : cx1 + Math.abs(cx2 - cx1) * 0.3;
              const cp1y = isVerticalJump ? cy1 + (cy2 - cy1) * 0.5 : cy1;
              const cp2x = isVerticalJump ? cx2 : cx2 - Math.abs(cx2 - cx1) * 0.3;
              const cp2y = isVerticalJump ? cy1 + (cy2 - cy1) * 0.5 : cy2;

              const isHighlighted = activeNodeIds && activeNodeIds.includes(edge.from) && activeNodeIds.includes(edge.to);
              const isFaded = activeNodeIds !== null && !isHighlighted;

              return (
                <path
                  key={edge.id}
                  d={`M ${cx1} ${cy1} C ${cp1x} ${cp1y}, ${cp2x} ${cp2y}, ${cx2} ${cy2}`}
                  fill="none" 
                  stroke={isHighlighted ? '#3b82f6' : (edge.dashed ? '#cbd5e1' : '#94a3b8')} 
                  strokeWidth={isHighlighted ? "5" : "3"} 
                  strokeDasharray={edge.dashed && !isHighlighted ? "8 8" : "none"}
                  markerEnd={isHighlighted ? "url(#arrowhead-highlight)" : "url(#arrowhead)"}
                  className={isHighlighted ? "edge-flow-fast" : (edge.dashed ? "" : "edge-flow")}
                  style={{ pointerEvents: 'none', transition: 'opacity 0.3s, stroke 0.3s', opacity: isFaded ? 0.1 : 0.8 }}
                />
              );
            })}

            {/* NODES RENDERER */}
            {activeView.nodes.map((node) => {
              const entity = schema.entities[node.id];
              if(!entity) return null;
              const type = entity.viewTypes[activeViewKey];
              if(!type) return null;

              const isSmall = [TYPES.USER, TYPES.EXTERNAL, TYPES.HOTSPOT, TYPES.DECISION].includes(type);
              const isHotspot = type === TYPES.HOTSPOT;
              const w = isSmall ? SMALL_W : NODE_W;
              const h = isSmall ? SMALL_H : NODE_H;
              const IconComponent = ICONS[type] || Activity;
              const animationClass = ICON_ANIMATIONS[type]; 
              
              const isSelected = selectedId === node.id && activeTab === 'entities';
              const isStepHighlighted = activeNodeIds && activeNodeIds.includes(node.id);
              const isDimmed = activeNodeIds !== null && !isStepHighlighted;
              
              return (
                <g 
                  key={node.id} 
                  className="canvas-node"
                  data-id={node.id}
                  transform={`translate(${node.x}, ${node.y})`}
                  style={{ 
                    cursor: draggingNode?.id === node.id ? 'grabbing' : 'grab',
                    transition: `opacity 0.5s ease-out ${mounted ? 0 : (node.x / 100) * 0.08}s, filter 0.3s`, 
                    opacity: mounted ? (isDimmed ? 0.25 : 1) : 0 
                  }}
                >
                  <rect 
                    x="0" y="0" width={w} height={h} 
                    fill={COLORS[type] || COLORS.default} stroke={BORDER_COLORS[type] || BORDER_COLORS.default} strokeWidth={isSelected || isStepHighlighted ? "4" : "2"} 
                    rx={type === TYPES.DATA_OBJECT ? "0" : (type === TYPES.DECISION ? "32" : "8")}
                    filter={isSelected || isStepHighlighted ? "url(#glow)" : "url(#shadow)"}
                  />
                  
                  <foreignObject x="0" y="0" width={w} height={h} style={{ pointerEvents: 'none' }}>
                    <div className="w-full h-full flex flex-col p-2.5 box-border select-none overflow-hidden">
                      <div className="flex items-center gap-1.5 pb-1 border-b border-black/10 mb-1 shrink-0">
                        <div className={`flex items-center justify-center ${mounted && !isDimmed ? animationClass : ''}`}>
                          <IconComponent size={14} color={BORDER_COLORS[type] || '#333'} />
                        </div>
                        <span className="text-[9px] uppercase font-bold tracking-wider truncate text-slate-800 opacity-70">
                          {type.split('/')[0]}
                        </span>
                      </div>
                      <div className="flex-1 flex items-center justify-center overflow-hidden">
                        <p className={`text-center font-bold text-slate-900 leading-tight ${isSmall ? 'text-[11px]' : 'text-[13px]'}`} style={{ display: '-webkit-box', WebkitLineClamp: isSmall ? 2 : 3, WebkitBoxOrient: 'vertical' }}>
                          {entity.title}
                        </p>
                      </div>
                    </div>
                  </foreignObject>
                </g>
              );
            })}
          </g>
        </svg>
      </div>

      {/* Editor Panel */}
      {renderSidebar()}
      
    </div>
  );
}
