How to use the CDN library

To render interactive sections on any static website, include the CSS library, the UMD Javascript bundle, and initialize sections using LoomSections.render(): 

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Loom Learning App</title>
  <!-- Load Loom Sections CSS Library -->
  <link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/loom-learning-sections@1.0.2/loom-sections.css">
</head>
<body style="background-color: #232634;">
  <!-- Container to render our interactive experience -->
  <div id="loom-root"></div>

  <!-- Load Loom Sections Javascript UMD Bundle -->
  <script src="https://cdn.jsdelivr.net/npm/loom-learning-sections@1.0.2/loom-sections.umd.js"></script>
  <script>
    const okfSections = [
      {
        type: "text",
        props: {
          title: "Introduction",
          paragraphs: [
            "# Welcome to Interactive Learning\nThis page loads sections via CDN!"
          ]
        }
      }
    ];
    
    // Render Loom sections into container
    LoomSections.render(document.getElementById("loom-root"), okfSections);
  </script>
</body>
</html>
